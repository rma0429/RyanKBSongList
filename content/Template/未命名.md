<%*
// 1. 基本資訊設定
const singerName = tp.file.title;
const dv = app.plugins.plugins.dataview.api;

// 2. 抓取並篩選頁面
const pages = dv.pages(`#Singer/${singerName}`)
    .filter(p => p.file.folder.includes("Ryan KB-Song List"))
    .sort(p => p.file.name, "asc");

const songCount = pages.length;

// 3. 生成標題
tR += "# 🎤 " + singerName + " 的所有歌曲 (共 " + songCount + " 首)\n\n";

// 4. 生成表格標頭 (確保欄位精準對齊)
tR += "| 歌名 | Key | 節奏 | BPM |\n";
tR += "| --- | --- | --- | --- |\n";

for (let p of pages) {
    // 【顯示優化】優先顯示屬性中的 title，若無則用檔名
    // 同時移除可能出現在 title 裡的 |，避免再次破壞表格
    let displayTitle = (p.title ? p.title : p.file.name).replace(/\|/g, " ");
    
    // 【路徑修復】精準模擬您提供的成功網址規則
    // 移除 .md、移除開頭的 content/、將「空格」替換為「-」
    let slugPath = p.file.path
        .replace(".md", "")
        .replace(/^content\//, "") 
        .replace(/ /g, "-");
    
    // 【連結生成】改用標準 Markdown 連結格式 [顯示標題](../路徑)
    // 這種格式不含 pipe (|)，絕對不會造成表格欄位錯位
    let songLink = `[${displayTitle}](../${slugPath})`;

    // 5. 抓取 Key/節奏/BPM 資料
    let key = p.file.etags
        .filter(t => t.startsWith(`#Singer/${singerName}`))
        .map(t => t.split("/")[2] || "-").join(", ");
    let styles = p.file.etags.filter(t => t.startsWith("#Style/"));
    let rhythm = styles.map(t => t.split("/")[1] || "-").join("<br>");
    let bpm = styles.map(t => t.split("/")[2] || "-").join("<br>");

    // 6. 輸出表格行
    tR += `| ${songLink} | ${key} | ${rhythm} | ${bpm} |\n`;
}
%>
