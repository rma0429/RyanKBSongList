<%*
// 1. 基本資訊
const singerName = tp.file.title;
const dv = app.plugins.plugins.dataview.api;

// 2. 獲取頁面
const pages = dv.pages(`#Singer/${singerName}`)
    .filter(p => p.file.folder.includes("Ryan KB-Song List"))
    .sort(p => p.file.name, "asc");

const songCount = pages.length;

// 3. 生成標題
tR += "# 🎤 " + singerName + " 的所有歌曲 (共 " + songCount + " 首)\n\n";

// 4. 生成表格標頭
tR += "| 歌名 | Key | 節奏 | BPM |\n";
tR += "| :--- | :--- | :--- | :--- |\n";

// --- 輔助函式區 ---

// 【核心修正】Quartz Slug 規則：
// 1. ⚠️ 絕對不要轉小寫 (因為您的網址保留了大小寫)
// 2. 空格變橫線
// 3. 移除特殊符號 (如括號、Pipe)
function toQuartzSlug(path) {
    let cleanPath = path.replace(".md", "").replace(/^content\//, "");
    
    let segments = cleanPath.split("/").map(s => {
        return s
                // .toLowerCase()  <-- 兇手就是這一行，我把它刪掉了！
                .replace(/\s+/g, "-")  // 空格轉橫線
                .replace(/[|()]/g, "") // 移除括號與 Pipe，避免破壞網址
                .replace(/\?|!/g, ""); // 移除問號驚嘆號
    });
    
    // 進行 URL 編碼 (確保中文路徑與空格安全)
    return encodeURI(segments.join("/"));
}

// 清理顯示文字
function cleanText(text) {
    if (!text) return "-";
    return text.toString()
        .replace(/\n/g, " ")   
        .replace(/\r/g, "")    
        .replace(/\|/g, "\\|") 
        .replace(/\[/g, "\\[") 
        .replace(/\]/g, "\\]");
}

let referenceLinks = [];

// --- 主迴圈 ---

pages.forEach((p, index) => {
    // 1. 顯示名稱
    let rawTitle = p.title ? p.title : p.file.name;
    let displayTitle = cleanText(rawTitle);
    
    // 2. 連結代號
    let linkId = `song-${index}`;
    
    // 3. 路徑計算
    // 根據網址結構 .../Singer/StephanieH (無斜線) -> ../Ryan... 應該就能到
    // 但為了保險起見，若 Quartz 內部有做資料夾轉向，../../ 比較穩
    // 這裡我們先用 ../../ (兩層)，如果跳轉到根目錄，代表您的設定是少見的「不轉資料夾」
    // 鑑於您之前的 404，我們先試試標準的 ../../ 配合「正確的大小寫」
    let relativeSlug = "../../" + toQuartzSlug(p.file.path);
    
    // 儲存實際連結
    referenceLinks.push(`[${linkId}]: ${relativeSlug}`);

    // 4. 表格內使用代號
    let songLink = `[${displayTitle}][${linkId}]`;

    // 5. 屬性提取
    let key = "-";
    if (p.file.etags) {
        let keyTag = p.file.etags.find(t => t.startsWith(`#Singer/${singerName}/`));
        if (keyTag) key = keyTag.split("/")[2] || "-";
    }

    let styles = p.file.etags ? p.file.etags.filter(t => t.startsWith("#Style/")) : [];
    let rhythm = styles.length > 0 ? styles.map(t => t.split("/")[1] || "-").join("<br>") : "-";
    let bpm = styles.length > 0 ? styles.map(t => t.split("/")[2] || "-").join("<br>") : "-";

    // 輸出
    tR += `| ${songLink} | ${cleanText(key)} | ${rhythm} | ${bpm} |\n`;
});

// 輸出隱藏的連結定義
tR += "\n\n" + referenceLinks.join("\n");
%>
